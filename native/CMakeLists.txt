cmake_minimum_required(VERSION 3.20)
project(BaroqueNative LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(RAYLIB_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../Simple 2000s DX  DECOMP/PingPong/build/toolchain/_deps/raylib-src"
    CACHE PATH "raylib 5.5 source tree (downloaded from GitHub when this folder does not exist)")
if(EXISTS "${RAYLIB_SOURCE}/src/raylib.h")
  add_subdirectory("${RAYLIB_SOURCE}" raylib)
else()
  include(FetchContent)
  FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz)
  FetchContent_MakeAvailable(raylib)
endif()
add_executable(Baroque-Native main.cpp)
file(GLOB BAROQUE_MODULE_CMAKE "${CMAKE_CURRENT_SOURCE_DIR}/cmake/*.cmake")
foreach(module_cmake ${BAROQUE_MODULE_CMAKE})
  include("${module_cmake}")
endforeach()
target_include_directories(Baroque-Native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(Baroque-Native PRIVATE raylib)
target_compile_definitions(Baroque-Native PRIVATE BAROQUE_ASSET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../work/assets")
if(MSVC)
  target_compile_options(Baroque-Native PRIVATE /W4 /O2 /utf-8)
endif()
